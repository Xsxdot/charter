### 1.3 设计裁决记录（plan 职责内的定稿，跨卡审计栏标待拍板）

- **D1｜diff 视图不进两轴**：选中 views 下拉的非 baseline 项时，内容区仍渲染基准快照并显示告知条「分支视图的两轴渲染未接入（二期）：当前显示的仍是基准快照」。依据：缝 1 六字段冻结输入（§2.3-19）没有 diff 维度，塞合成视图运行期即炸（§0 实测）；CgView 对象边也无法安全转回元组而不丢 flows/implements（丢了好过假装能渲染？都不行——所以如实告知）。旧世界的对应行为是「分支视图暂用现状域全景——per-view 对照见二期」（CodegraphPage.tsx :317-320 原 banner），本裁决是该同一搁置在新架构下的直系后代，不是功能回退的新决定。
- **D2｜工具条保留四件**：视图下拉、失鲜徽标、单域提示 pill、刷新按钮原样保留（含 NOT_SCANNED 占位三分态）。它们是数据层关注点不属于任何轴，spec/contract 的退役名单均不含它们；K6 验收 3「views/stale/report 缺失等旧降级分支行为保持」由此落地。单域提示文案从「重扫可获得领域全景」改为「重扫可获得两轴全景」（世界变了，提示跟着变；降级语义不变）。hasDomains 不再可用（D3），判空就地内联为 `Object.keys(view.domains ?? {}).length === 0`。
- **D3｜P5-A 删除集按枚举扩大到 33 文件**：见 §0 枚举表。原则只有一条——装配切换后生产 importer 归零即删，besttree 因四个活消费方整文件保留。
- **D4｜entryName 边界**：只动 `FlowPageModel` 输出侧十键中的新键与 return 处一行；FlowPageInput、deriveFlowPage 地址、九个旧键的语义与顺序无关判断全部不动（键集断言由 T1 同步从九键改十键，属配套断言非行为变更）。幽灵入口输出空串，视图层 id 兜底——模型不猜名字（沿 scopepage「悬空端点整条忽略」同一诚实纪律）。
- **D5｜换视图整页重挂**：`<TwoAxisPage key={viewName}>`——scope/组织/抽屉是上一层视图的语境，带过去只会误导（沿旧 goScope 清空先例）；行为轴↔结构轴切换则是条件渲染，回到结构轴回到根层（层级语境不作跨轴保留，同款先例）。

// classname-drift —— §2.5-40 防漂判据的机械检查：两轴新世界的交互控件必须有非空 className。
//
// 职责：对装配切换后仍存活的七个组件源码做静态扫描——每个 <button>/<select>/<a>/

import { join } from 'node:path'
import { describe, expect, it } from 'vitest'

  'TwoAxisPage.tsx',

  'RightPanel.tsx',
  'MigrationDrawer.tsx',
  'FlowChart.tsx',
  'FlowPageView.tsx',
  'CodegraphPage.tsx',
] as const

const INTERACTIVE_TAGS = ['button', 'select', 'a', 'input', 'textarea'] as const

/** 从标签起点按引号/花括号平衡找真正的 '>'（JSX 属性里允许 =>、对象字面量、多行）。 */
function tagEndAt(src: string, start: number): number {
  let i = start
  let quote = ''
  while (i < src.length) {
    const c = src[i]!
    if (quote !== '') {
      if (c === quote) quote = ''
    } else if (c === '"' || c === "'") {
      quote = c
    } else if (c === '{') {
      let depth = 1
      i += 1
      while (i < src.length && depth > 0) {
        if (src[i] === '{') depth += 1
        else if (src[i] === '}') depth -= 1
        i += 1
      }
      continue
    } else if (c === '>') {
      return i
    }
    i += 1
  }
  return -1
}

function findViolations(source: string): string[] {
  const violations: string[] = []
  const tagOpen = new RegExp(`<(${INTERACTIVE_TAGS.join('|')})[\\s>]`, 'g')
  for (const m of source.matchAll(tagOpen)) {
    const end = tagEndAt(source, m.index)
    if (end < 0) continue
    const tag = source.slice(m.index, end + 1)
    const line = source.slice(0, m.index).split('\n').length
    const cm = tag.match(/className\s*=\s*("([^"]*)"|'([^']*)'|\{)/)
    if (!cm) {
      violations.push(`line ${line}: <${m[1]}> 缺 className`)
      continue
    }
    const literal = cm[2] ?? cm[3]
    if (literal !== undefined && literal.trim() === '') {
      violations.push(`line ${line}: <${m[1]}> className 为空白字面量`)
    }
  }
  return violations
}

describe('C12.6 §2.5-40 防漂：交互控件非空 className 机械检查', () => {
  for (const file of SCANNED_COMPONENTS) {
    it(`${file} 全部交互控件带非空 className`, () => {
      const source = readFileSync(join(CODEGRAPH_DIR, file), 'utf8')
      expect(findViolations(source)).toEqual([])
    })
  }

  it('扫描器自身有牙：构造缺 className 的样本必须被抓到', () => {
    expect(findViolations('<button type="button" data-x>\n点我</button>')).toEqual([
      'line 1: <button> 缺 className',
    ])
    expect(findViolations('<select onChange={() => f(">")} className="">')).toEqual([
      'line 1: <select> className 为空白字面量',
    ])
    expect(findViolations("<a href=\"#x\" className={'rounded ' + active}>链接</a>")).toEqual([])
  })
